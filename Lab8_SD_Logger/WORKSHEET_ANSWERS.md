### ข้อ 1 — ช่วงบันทึกและขนาดไฟล์

constexpr uint32_t LOG_INTERVAL_MS = 2000;

### ข้อ 2 — คอลัมน์ `wifi_rssi`
```cpp
// createLogFile():
file.println("timestamp,uptime_s,source,temp_c,hum_pct,fan_duty_pct,mode,wifi_rssi");

// appendLogRow():
char rssiText[8];
if (wifiState == WifiState::Connected) snprintf(rssiText, sizeof(rssiText), "%d", WiFi.RSSI());
else snprintf(rssiText, sizeof(rssiText), "NA");
snprintf(line, sizeof(line), "%s,%lu,%s,%.1f,%.0f,%d,%s,%s", stamp, ..., mode == Mode::Auto ? "AUTO" : "MANUAL", rssiText);
```
(build ผ่านแล้ว) ผลที่คาด: 8 คอลัมน์ แถวที่เชื่อมต่ออยู่ เช่น `...,AUTO,-52` ขนาดแถวใหม่ = 44 + 4 = **48 ไบต์** (`,-52` 4 ตัว) หรือ 44 + 3 = 47 เมื่อเป็น `,NA` และหัวตาราง 60 + 10 (`,wifi_rssi`) = 70 ไบต์ (ผลบนบอร์ดยังไม่ได้ทดสอบ) ต้องแก้ทั้งหัวตารางและแถว มิฉะนั้นจำนวนคอลัมน์ไม่ตรงกัน

### ข้อ 3 — รอ sync ก่อนบันทึก
```cpp
if (sdState != SdState::Ready || !timeValid()) {
    logSkipped++;
    return;
}
```
(build ผ่านแล้ว) เปรียบเทียบ: ข้อดี ทุกแถวมีเวลาจริง ไฟล์สะอาด ข้อเสีย เสียข้อมูลช่วงก่อน sync และถ้า **ไม่มี WiFi เลย จะไม่บันทึกแม้แต่แถวเดียว** (`rows` ค้างที่ 0, `skipped` เพิ่มเรื่อยๆ) ซึ่งไม่เหมาะกับ logger ที่ต้องทำงานแม้ไม่มีอินเทอร์เน็ต วิธี `NOT_SYNCED` (หัวข้อ 3.8) จึงเป็นทางเลือกที่ยืดหยุ่นกว่า (ผลบนบอร์ดยังไม่ได้ทดสอบ)

### ข้อ 4 — ลดการสะดุดเมื่อไม่มีการ์ด
```cpp
constexpr uint32_t SD_RETRY_MS = 30000;
```
(build ผ่านแล้ว) สัดส่วนที่ถูกบล็อก: 625 ÷ (30,000 + 625) ≈ **2.0%** (เทียบกับ ≈ 11% เดิม) ข้อเสีย: เมื่อใส่การ์ดแล้วอาจต้องรอนานถึง ≈ 30 วินาทีกว่าจะเมาท์และบันทึกต่อ (หรือพิมพ์ `r` เพื่อสั่งทันที) ผลที่เห็นบนจอยังไม่ได้ทดสอบ

### ข้อ 5 — ความเร็ว SPI

```cpp
constexpr uint32_t SD_SPI_HZ = 10000000;   // แล้วลอง 20000000
```
**คำทำนายที่ถูก:** ลดลง **ไม่ถึง 5 เท่า** เพราะเวลาเขียนส่วนใหญ่ไม่ใช่เวลาส่งข้อมูลบน SPI (เวลาส่ง 44 ไบต์ที่ 4 MHz = 88 µs, 512 ไบต์ ≈ 1 ms) แต่เป็นงานของระบบไฟล์ (เปิด/ปิดไฟล์, อัปเดต FAT) และการรอการ์ด

**ผลที่วัดได้จริงบนการ์ดตัวนี้** (ตัวอย่างเล็ก การ์ดตัวเดียว ไม่ใช่ค่าที่รับประกันทุกการ์ด)

| ความเร็ว SPI | เมาท์ | เวลาเขียน (`write took`) | หมายเหตุ |
|---|---|---|---|
| 4 MHz (ค่าเริ่มต้น) | 10–148 ms | 14,372–16,838 µs (≈ 14.4–16.8 ms) | หลายครั้ง |
| 10 MHz | 8 ms | 14,164 / 12,449 / **30,617** µs | แถวที่ 3 เป็นค่าสูงผิดปกติ 1 ครั้ง |
| 20 MHz | 7 ms | 13,238 / 11,701 / 11,857 µs (≈ 11.7–13.2 ms) | เมาท์ผ่าน เขียนได้ปกติ |

สรุป: เพิ่มความเร็ว SPI 5 เท่า (4 → 20 MHz) ลดเวลาเขียนลงเพียงประมาณ 20–30% (จาก ≈ 14–17 ms เหลือ ≈ 12–13 ms) เมาท์ผ่านทั้งสองความเร็วบนการ์ดนี้ แต่การ์ด/สายอื่นอาจไม่ผ่าน และค่าเริ่มต้น 4 MHz ปลอดภัยกว่า ผลจากการเขียนทุก 10 วินาทีไม่ต่างกันในทางปฏิบัติ (0.16% เทียบ ~0.12%)

(ระหว่างทดสอบ 20 MHz รอบหนึ่ง สองแถวแรกเป็น `NOT_SYNCED` เพราะ sync เวลาเสร็จช้ากว่าปกติในบูตนั้น ไม่เกี่ยวกับความเร็ว SPI แต่เป็นตัวอย่างที่ดีของเหตุผลที่ต้องมีวิธีรับมือ `NOT_SYNCED`)

### ข้อ 6 — ไฟล์ใหม่ทุกวัน
```cpp
long lastLogDay = -1;   // ตัวแปรร่วม

// ใน appendLogRow() ก่อนเขียน:
if (timeValid()) {
    const long day = (static_cast<long>(time(nullptr)) + app::GMT_OFFSET_SEC + app::DAYLIGHT_OFFSET_SEC) / 86400;
    if (lastLogDay >= 0 && day != lastLogDay) {
        createLogFile();     // วันใหม่ -> ไฟล์ใหม่
    }
    lastLogDay = day;
}
```
(build ผ่านแล้ว) ต้องใช้เวลาท้องถิ่น (บวก `GMT_OFFSET_SEC`) ไม่เช่นนั้นไฟล์จะเปลี่ยนตอน 07:00 น. เวลาไทย (ข้อผิดพลาดแบบเดียวกับที่เคยเจอใน Lab 7 กับวันที่บนจอ) วิธีทดสอบโดยไม่รอเที่ยงคืน: ตั้ง `GMT_OFFSET_SEC` ชั่วคราวให้เวลาท้องถิ่นใกล้เที่ยงคืน (เช่นตอนนี้ 20:00 UTC+7 ตั้ง offset เป็น 11 ชั่วโมงจะเป็น 00:00 ของวันถัดไป) หรือจำลองเลขวันด้วยตัวนับ (ผลบนบอร์ดยังไม่ได้ทดสอบ)

### ข้อ 7 — สถิติจากไฟล์ (คำสั่ง `a`)
```cpp
void printLogStats()
{
    if (sdState != SdState::Ready || logPath[0] == '\0') { Serial.println("[SD] no log file is open"); return; }
    File file = SD.open(logPath, FILE_READ);
    if (!file) { Serial.println("[SD] cannot open the log file"); return; }
    char line[96];
    size_t length = 0;
    bool header = true;
    int rows = 0;
    double sum = 0.0;
    float minTemp = 1000.0f, maxTemp = -1000.0f;
    while (file.available()) {
        const int c = file.read();
        if (c == '\n') {
            line[length] = '\0';
            length = 0;
            if (header) { header = false; continue; }          // ข้ามหัวตาราง
            const char *field = line;
            for (int i = 0; i < 3 && field; i++) {              // ผ่าน 3 คอมมาแรก
                field = strchr(field, ',');
                if (field) field++;
            }
            if (field) {
                const float value = static_cast<float>(atof(field));
                sum += value;
                if (value < minTemp) minTemp = value;
                if (value > maxTemp) maxTemp = value;
                rows++;
            }
        } else if (c != '\r' && length < sizeof(line) - 1) {
            line[length++] = static_cast<char>(c);
        }
    }
    file.close();
    if (rows == 0) Serial.println("[STATS] no data rows yet");
    else Serial.printf("[STATS] %d rows: min=%.1f max=%.1f avg=%.1f C\n", rows, minTemp, maxTemp, sum / rows);
}
// ใน handleSerialInput(): else if (c == 'a' || c == 'A') printLogStats();
```
**ตรวจบนบอร์ดจริงแล้ว:** `[STATS] 3 rows: min=29.7 max=29.8 avg=29.8 C` (ไฟล์มี 3 แถว: 29.8, 29.8, 29.7 → เฉลี่ย 29.77 ปัดเป็น 29.8) ข้อสังเกต: อ่านทีละไบต์ไม่โหลดทั้งไฟล์ลง RAM, `atof` อ่านตัวเลขจนถึงคอมมาถัดไป, ข้ามตัว `\r` ของ Windows

## ข้อควรระวังและข้อสังเกตสำหรับผู้สอน

1. **ใช้การ์ดทดลอง** อย่าใช้การ์ดที่มีข้อมูลสำคัญ แม้โปรแกรมไม่ฟอร์แมตและลบเฉพาะ `log_####.csv` แต่การถอดผิดวิธีอาจทำให้ไฟล์ระบบเสียหาย
2. **ถอดการ์ดตอนปิดไฟหรือหลังสั่ง `e` เท่านั้น** ย้ำให้นักเรียนทำเป็นนิสัย
3. **ผลวัดเวลาเป็นของการ์ดตัวเดียว** การ์ดแต่ละยี่ห้อ/รุ่นเขียนช้าเร็วต่างกัน (และอาจมีค่าสูงผิดปกติเป็นครั้งคราว เช่น 30 ms ที่เจอครั้งหนึ่ง) ให้ตรวจแนวโน้ม ไม่ใช่ตัวเลขเป๊ะ
4. **การ์ด ≥ 32 GB** มักเป็น exFAT ซึ่งยังไม่ได้ทดสอบ ให้ฟอร์แมตเป็น FAT32 ก่อน
5. **ข้อ 5 (ความเร็ว SPI):** ผลขึ้นกับการ์ดและสายต่อ (บอร์ดนี้ใช้สล็อตบนแผ่นวงจรเดียวกัน จึงเสถียรกว่าสายยาว) ถ้าเมาท์ไม่ผ่านที่ความเร็วสูง ให้ย้อนกลับ 4 MHz ไม่ถือเป็นข้อผิดพลาด
6. **ข้อ 6:** ย้ำเรื่องเวลาท้องถิ่น vs UTC (เชื่อมโยงกับข้อบกพร่องวันที่ใน Lab 7)
7. **ไฟล์ทดสอบที่เหลือบนการ์ด** ให้นักเรียนลบด้วย `x`/`y` หรือจากคอมพิวเตอร์เมื่อเสร็จ
